IIFE com jQuery

 (function($) {
  /* Código jQuery aqui */
  $(document).ready(function() {
    console.log("Rodando IIFE, documento pronto e jQuery carregado!");
  });
 })(jQuery);

 /* Com arrow function (se quiser deixar mais compacta) */ 
 (($) => {
  $(function() {
    console.log("Rodando com jQuery em IIFE");
  });
 })(jQuery);

 /* Immediately Invoked Function Expression - IIFE */
