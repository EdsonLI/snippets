<!-- Estrutura HTML para Isotope -->
<div class="controls">
  <button data-filter="*">Todos</button>
  <button data-filter=".categoria-1">Categoria 1</button>
  <button data-filter=".categoria-2">Categoria 2</button>
</div>
<div class="grid">
  <div class="grid-item categoria-1" data-name="Alpha" data-number="3">Alpha - 3</div>
  <div class="grid-item categoria-2" data-name="Beta" data-number="1">Beta - 1</div>
  <div class="grid-item categoria-1" data-name="Gamma" data-number="2">Gamma - 2</div>
</div>

        

.grid {
  display: flex;
  flex-wrap: wrap;
  gap: 1rem;
}
.grid-item {
  background: #222;
  color: #5e65cc;
  padding: 1rem;
  border-radius: 5px;
  width: 120px;
}
.controls button {
  margin-right: 0.5rem;
  padding: 0.4rem 1rem;
  cursor: pointer;
}
.controls button.active {
  background: #5e65cc;
  color: black;
}



$(function() {
  var $grid = $('.grid').isotope({
    itemSelector: '.grid-item',
    layoutMode: 'fitRows',
  });

  $('.controls').on('click', 'button', function() {
    $('.controls button').removeClass('active');
    $(this).addClass('active');
    var filterValue = $(this).attr('data-filter');
    $grid.isotope({ filter: filterValue });
  });
});
