Cookies, Sessões e Preferências

<?php
  session_start();

  # Aceite de cookies recebido via POST
  if (isset($_POST['aceitar'])) {
    setcookie('aceite_cookie', '1', time() + (86400 * 30), '/'); // 30 dias
    $_SESSION['tema'] = 'escuro'; // Exemplo de preferência
  }

  # Mostrar sessão e cookie (debug/uso)
  if (isset($_COOKIE['aceite_cookie'])) {
    echo "Cookie aceite_cookie: " . $_COOKIE['aceite_cookie'] . "<br>";
  }

  if (isset($_SESSION['tema'])) {
      echo "Sessão tema: " . $_SESSION['tema'] . "<br>";
  }
?>

<!-- Mostrar aviso se cookie ainda não aceito -->
<?php if (!isset($_COOKIE['aceite_cookie'])): ?>
   <form method="post" style="position: fixed; bottom: 1rem; left: 1rem; right: 1rem; background: #222; color: #fff; padding: 1rem; border-radius: .5rem; z-index: 1000;">
      Este site usa cookies para salvar suas preferências.
      <button type="submit" name="aceitar" class="btn btn-sm btn-success ms-2">Aceitar</button>
   </form>
<?php endif; ?>
