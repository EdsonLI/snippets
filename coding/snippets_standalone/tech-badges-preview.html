SQL
Procedure para gerar relatório simples
DELIMITER //

CREATE PROCEDURE Relatorio_Clientes_Por_Cidade()
BEGIN
  SELECT cidade, COUNT(*) AS total_clientes
    FROM clientes
  GROUP BY cidade;
END //

DELIMITER ;
