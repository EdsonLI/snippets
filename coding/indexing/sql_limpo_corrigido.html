Create Database
CREATE DATABASE loja;

USE loja;

-- Ou criar e usar
CREATE DATABASE IF NOT EXISTS loja 
CHARACTER SET utf8mb4 
COLLATE utf8mb4_unicode_ci;
Create Table
CREATE TABLE usuarios (
    id INT AUTO_INCREMENT PRIMARY KEY,
    nome VARCHAR(100) NOT NULL,
    email VARCHAR(255) UNIQUE,
    senha VARCHAR(255),
    data_criacao TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    ativo BOOLEAN DEFAULT TRUE
);
Insert Into
-- Inserir um registro
INSERT INTO usuarios (nome, email, senha) 
VALUES ('João Silva', 'joao@email.com', 'senha123');

-- Inserir múltiplos registros
INSERT INTO usuarios (nome, email, senha) 
VALUES 
    ('Maria Santos', 'maria@email.com', 'pass456'),
    ('Pedro Oliveira', 'pedro@email.com', 'abc789');
Select Básico
-- Selecionar todos os registros
SELECT * FROM usuarios;

-- Selecionar campos específicos
SELECT nome, email FROM usuarios;

-- Com condição WHERE
SELECT * FROM usuarios 
WHERE ativo = TRUE;
Update
-- Atualizar um registro
UPDATE usuarios 
SET email = 'novo@email.com' 
WHERE id = 1;

-- Atualizar múltiplos campos
UPDATE usuarios 
SET nome = 'João Santos', ativo = FALSE 
WHERE email = 'joao@email.com';
Delete
-- Deletar registro específico
DELETE FROM usuarios 
WHERE id = 1;

-- Deletar com múltiplas condições
DELETE FROM usuarios 
WHERE ativo = FALSE 
AND data_criacao < '2024-01-01';
Join
-- INNER JOIN
SELECT u.nome, p.titulo
FROM usuarios u
INNER JOIN posts p ON u.id = p.usuario_id;

-- LEFT JOIN
SELECT u.nome, COUNT(p.id) as total_posts
FROM usuarios u
LEFT JOIN posts p ON u.id = p.usuario_id
GROUP BY u.id, u.nome;
Where Avançado
-- Operadores de comparação
SELECT * FROM produtos 
WHERE preco BETWEEN 10 AND 50;

-- LIKE para busca por padrão
SELECT * FROM usuarios 
WHERE nome LIKE 'João%';

-- IN para múltiplos valores
SELECT * FROM produtos 
WHERE categoria_id IN (1, 3, 5);
Order By & Limit
-- Ordenar por campo
SELECT * FROM usuarios 
ORDER BY nome ASC;

-- Ordenar por múltiplos campos
SELECT * FROM produtos 
ORDER BY categoria_id DESC, preco ASC;

-- Limitar resultados
SELECT * FROM posts 
ORDER BY data_criacao DESC 
LIMIT 10;
Group By
-- Agrupar e contar
SELECT categoria_id, COUNT(*) as total
FROM produtos 
GROUP BY categoria_id;

-- Agrupar com HAVING
SELECT categoria_id, AVG(preco) as preco_medio
FROM produtos 
GROUP BY categoria_id
HAVING AVG(preco) > 100;
Alter Table
-- Adicionar coluna
ALTER TABLE usuarios 
ADD COLUMN telefone VARCHAR(20);

-- Modificar coluna
ALTER TABLE usuarios 
MODIFY COLUMN nome VARCHAR(150);

-- Remover coluna
ALTER TABLE usuarios 
DROP COLUMN telefone;
Índices
-- Criar índice
CREATE INDEX idx_usuario_email 
ON usuarios (email);

-- Índice composto
CREATE INDEX idx_produto_categoria_preco 
ON produtos (categoria_id, preco);

-- Remover índice
DROP INDEX idx_usuario_email 
ON usuarios;
