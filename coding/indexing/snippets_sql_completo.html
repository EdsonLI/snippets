CREATE TABLE clientes (
  id INT AUTO_INCREMENT PRIMARY KEY,
  nome VARCHAR(100) NOT NULL,
  email VARCHAR(100) UNIQUE,
  telefone VARCHAR(20),
  data_nascimento DATE,
  data_cadastro TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

Funções Agregadas 
-- Contagem
SELECT COUNT(*) AS total FROM clientes;

-- Soma
SELECT SUM(valor) AS total FROM pedidos;

-- Média
SELECT AVG(valor) AS media FROM pedidos;

-- Valor máximo
SELECT MAX(valor) AS maior_pedido FROM pedidos;

-- Valor mínimo
SELECT MIN(valor) AS menor_pedido FROM pedidos;
GROUP BY 
-- Agrupar com contagem
SELECT cidade, COUNT(*) AS total
FROM clientes

-- Agrupar com condição
SELECT cidade, COUNT(*) AS total
FROM clientes
GROUP BY cidade
HAVING COUNT(*) > 5
ORDER BY total DESC;
