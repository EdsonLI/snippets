Conexão MySQL 
<?php
$host = "localhost";
$dbname = "minha_db";
$username = "root";
$password = "";

try {
    $pdo = new PDO(
        "mysql:host=$host;dbname=$dbname",
        $username, 
        $password
    );
    $pdo->setAttribute(
        PDO::ATTR_ERRMODE, 
        PDO::ERRMODE_EXCEPTION
    );
} catch(PDOException $e) {
    echo "Erro: " . $e->getMessage();
}
?>
SELECT com PDO 
<?php
// Buscar todos

// Buscar com parâmetros

    "SELECT * FROM users WHERE age > ?"
);
$stmt->execute([18]);
$adults = $stmt->fetchAll();
?>
INSERT com PDO 
<?php
$stmt = $pdo->prepare(
    "INSERT INTO users (name, email, age) 
     VALUES (?, ?, ?)"
);

$stmt->execute([
    "João Silva",
    "joao@email.com",
    30
]);

echo "Usuário inserido com ID: " . 
     $pdo->lastInsertId();
?>
Try/Catch 
<?php
try {
    $result = 10 / 0;
} catch (DivisionByZeroError $e) {
    echo "Erro: Divisão por zero!";
} catch (Exception $e) {
    echo "Erro geral: " . $e->getMessage();
} finally {
    echo "Sempre executa";
}
?>
Classes 
<?php
class Person {
    public $name;
    private $age;
    
    public function __construct($name, $age) {
        $this->name = $name;
        $this->age = $age;
    }
    
    public function getAge() {
        return $this->age;
    }
}

$person = new Person("Maria", 25);
echo $person->name;
echo $person->getAge();
?>
