FUNÇÕES 
function saudacao($nome) {
    return "Olá, " . $nome . "!";
}

function calcular($a, $b, $operacao = '+') {
    switch ($operacao) {
        case '+': return $a + $b;
        case '-': return $a - $b;
        case '*': return $a * $b;
        case '/': return $b != 0 ? $a / $b : 0;
        default: return 0;
    }
}

echo saudacao("João");
echo calcular(10, 5, '*');
CLASSES 
class Usuario {
    private $nome;
    private $email;
    
    public function __construct($nome, $email) {
        $this->nome = $nome;
        $this->email = $email;
    }
    
    public function getNome() {
        return $this->nome;
    }
    
    public function exibirInfo() {
        return $this->nome . " - " . $this->email;
    }
}

$user = new Usuario("Ana", "ana@email.com");
echo $user->exibirInfo();

    $pdo = new PDO('mysql:host=localhost;dbname=test', $user, $pass);
    $pdo->setAttribute(PDO::ATTR_ERRMODE, PDO::ERRMODE_EXCEPTION);
    
    # Inserir
    $stmt = $pdo->prepare("INSERT INTO usuarios (nome, email) VALUES (?, ?)");
    $stmt->execute(['João', 'joao@email.com']);
    
    # Selecionar
    $stmt = $pdo->prepare("SELECT * FROM usuarios WHERE ativo = ?");
    $stmt->execute([1]);
    $usuarios = $stmt->fetchAll(PDO::FETCH_ASSOC);
    
    foreach ($usuarios as $usuario) {
        echo $usuario['nome'] . "\n";
    }
} catch (PDOException $e) {
    echo "Erro: " . $e->getMessage();

# Função personalizada de erro
function meuErro($errno, $errstr, $errfile, $errline) {
    error_log("Erro [$errno] $errstr - Arquivo: $errfile, Linha: $errline");
    return true;
}

set_error_handler("meuErro");
