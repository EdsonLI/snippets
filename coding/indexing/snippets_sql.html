CREATE TABLE
CREATE TABLE usuarios (
    id INT AUTO_INCREMENT PRIMARY KEY,
    nome VARCHAR(100) NOT NULL,
    email VARCHAR(150) UNIQUE NOT NULL,
    idade INT,
    data_criacao TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- Múltiplas condições
SELECT * FROM usuarios 
WHERE idade > 20 AND nome LIKE 'J%';

SELECT * FROM usuarios 
WHERE idade IN (25, 30, 35);

-- BETWEEN
SELECT * FROM usuarios

UPDATE

UPDATE usuarios 
SET idade = 31 
WHERE id = 1;

-- Atualizar múltiplos campos
UPDATE usuarios

WHERE id = 1;

SELECT u.nome, p.total, pr.nome as produto
FROM usuarios u
INNER JOIN pedidos p ON u.id = p.usuario_id
INNER JOIN produtos pr ON p.produto_id = pr.id;
LEFT JOIN
-- Todos os usuários, com ou sem pedidos
SELECT u.nome, p.total
FROM usuarios u
LEFT JOIN pedidos p ON u.id = p.usuario_id;

-- Usuários sem pedidos
SELECT u.nome
FROM usuarios u
LEFT JOIN pedidos p ON u.id = p.usuario_id
WHERE p.id IS NULL;
GROUP BY
-- Contar usuários por idade
SELECT idade, COUNT(*) as total
FROM usuarios
GROUP BY idade;

-- Com HAVING
SELECT idade, COUNT(*) as total
FROM usuarios
GROUP BY idade
HAVING COUNT(*) > 1;
Funções Agregadas
SELECT 
    COUNT(*) as total_usuarios,
    AVG(idade) as idade_media,
    MIN(idade) as menor_idade,
    MAX(idade) as maior_idade,
    SUM(idade) as soma_idades
FROM usuarios;
ORDER BY
-- Ordenar por nome (A-Z)
SELECT * FROM usuarios 
ORDER BY nome ASC;

-- Ordenar por idade (maior primeiro)
SELECT * FROM usuarios 
ORDER BY idade DESC;

-- Múltiplos critérios
SELECT * FROM usuarios 
ORDER BY idade DESC, nome ASC;
LIMIT 
-- Primeiros 5 registros
SELECT * FROM usuarios 
LIMIT 5;

-- Paginação (pular 10, pegar 5)
SELECT * FROM usuarios 
ORDER BY nome
LIMIT 10, 5;

-- Equivalente com OFFSET
SELECT * FROM usuarios 
ORDER BY nome
LIMIT 5 OFFSET 10;

Subqueries 
-- Usuários com idade acima da média
SELECT nome, idade FROM usuarios
WHERE idade > (
    SELECT AVG(idade) FROM usuarios
);

-- EXISTS
SELECT nome FROM usuarios u
WHERE EXISTS (
    SELECT 1 FROM pedidos p 
    WHERE p.usuario_id = u.id
);
CASE WHEN

    CASE 
        WHEN idade < 18 THEN 'Menor'
        WHEN idade BETWEEN 18 AND 59 THEN 'Adulto'
        ELSE 'Idoso'
    END as categoria
FROM usuarios;
