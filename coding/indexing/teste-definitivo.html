        apiUrl: "https://api.exemplo.com",
        timeout: 5000,
        debug: true
    };
    
    // Função assíncrona
    async function buscarDados(endpoint) {
        try {
            const response = await fetch(config.apiUrl + endpoint, {
                method: 'GET',
                headers: {
                    'Content-Type': 'application/json',
                    'Authorization': 'Bearer ' + token
                }
            });
            
            if (!response.ok) {
                throw new Error(`Erro ${response.status}`);
            }
            
            return await response.json();
            
        } catch (error) {
            console.error("Erro:", error.message);
            return null;
        }
    }
    
    // Event listener
    $('.btn-carregar').on('click', function(e) {
        e.preventDefault();
        
        const $btn = $(this);
        $btn.prop('disabled', true);
        
        buscarDados('/usuarios').then(dados => {
            $('#resultado').html(JSON.stringify(dados, null, 2));
        }).finally(() => {
            $btn.prop('disabled', false);
        });
    });
});

-- Query com JOINS e agregações
SELECT 
    u.id,
    u.nome,
    u.email,
    COUNT(p.id) as total_pedidos,
    SUM(p.valor_total) as valor_gasto,
    AVG(p.valor_total) as ticket_medio
FROM usuarios u
LEFT JOIN pedidos p ON u.id = p.usuario_id
WHERE u.ativo = TRUE
  AND u.created_at >= DATE_SUB(NOW(), INTERVAL 6 MONTH)
GROUP BY u.id, u.nome, u.email
HAVING total_pedidos > 5
ORDER BY valor_gasto DESC
LIMIT 100;

-- Criação de índice
CREATE INDEX idx_usuario_ativo ON usuarios(ativo, created_at);
