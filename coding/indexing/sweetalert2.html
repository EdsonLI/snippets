Select com Validação
Swal.fire({
  title: 'Escolha uma opção',
  input: 'select',
  inputOptions: {
    'opcao1': 'Opção 1',
    'opcao2': 'Opção 2',  
    'opcao3': 'Opção 3'
  },
  inputPlaceholder: 'Selecione uma opção',
  showCancelButton: true,
  confirmButtonText: 'Confirmar',
  cancelButtonText: 'Cancelar',
  inputValidator: (value) => {
    return !value && 'Você precisa selecionar uma opção!'
  }
}).then((result) => {
  if (result.isConfirmed) {
    Swal.fire('Sucesso!', `Você escolheu: ${result.value}`, 'success');
  }
});

Input de Texto
Swal.fire({
  title: 'Digite seu nome',
  input: 'text',
  inputPlaceholder: 'Seu nome aqui...',
  showCancelButton: true,
  inputValidator: (value) => {
    if (!value) {
      return 'Você precisa digitar algo!'
    }
    if (value.length < 3) {
      return 'Nome deve ter pelo menos 3 caracteres'
    }
  }
}).then((result) => {
  if (result.isConfirmed) {
    Swal.fire(`Olá, ${result.value}!`);
  }
});

// Com contador regressivo
let timerInterval
Swal.fire({
  title: 'Redirecionando...',
  html: 'Será redirecionado em  segundos.',
  timer: 5000,
  timerProgressBar: true,
  didOpen: () => {
    const b = Swal.getHtmlContainer().querySelector('b')
    timerInterval = setInterval(() => {
      b.textContent = Math.ceil(Swal.getTimerLeft() / 1000)
    }, 100)
  },
  willClose: () => {
    clearInterval(timerInterval)
  }
});

// Com simulação de processo
Swal.fire({
  title: 'Processando...',
  html: 'Progresso: 0%',
  showConfirmButton: false,
  allowOutsideClick: false,
  didOpen: () => {
    Swal.showLoading()
    const b = Swal.getHtmlContainer().querySelector('b')
    let progress = 0
    const timer = setInterval(() => {
      progress += 10
      b.textContent = progress + '%'
      if (progress >= 100) {
        clearInterval(timer)
        Swal.fire('Concluído!', 'Processo finalizado', 'success')
      }
    }, 300)
  }
});
